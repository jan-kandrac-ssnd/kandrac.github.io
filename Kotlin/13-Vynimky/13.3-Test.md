<div class="quiz-item active" data-correct="yes">

  </div>

</div>

<div class="quiz-item" data-correct="yes">
  <p class="quiz-question">Jeden blok <code>try</code> môže mať viacero <code>catch</code> vetiev pre rôzne typy výnimiek.</p>
  <div class="quiz-actions">
    <button type="button" class="quiz-btn quiz-yes">Áno</button>
    <button type="button" class="quiz-btn quiz-no">Nie</button>

  <p class="quiz-comment"><strong>Áno.</strong> Napríklad môžeme mať samostatný <code>catch (e: ArithmeticException)</code> a <code>catch (e: NumberFormatException)</code>.</p>

<div class="quiz-item" data-correct="yes">
  <p class="quiz-question">Blok <code>finally</code> sa vykoná vždy - bez ohľadu na to, či v <code>try</code> nastala výnimka alebo nie.</p>
  <div class="quiz-actions">
    <button type="button" class="quiz-btn quiz-yes">Áno</button>
    <button type="button" class="quiz-btn quiz-no">Nie</button>
  </div>
  <p class="quiz-comment"><strong>Áno.</strong> <code>finally</code> je určený na kód, ktorý sa musí vykonať vždy, napríklad upratanie zdrojov.</p>
</div>

<div class="quiz-item" data-correct="yes">
  <p class="quiz-question"><code>try-catch</code> môže byť v Kotline použitý ako výraz a vrátiť hodnotu, ktorá sa uloží do premennej.</p>
  <div class="quiz-actions">
    <button type="button" class="quiz-btn quiz-yes">Áno</button>
    <button type="button" class="quiz-btn quiz-no">Nie</button>
  </div>
  <p class="quiz-comment"><strong>Áno.</strong> Napríklad <code>val cislo = try { readln().toInt() } catch (e: NumberFormatException) { null }</code>.</p>
</div>

<div class="quiz-item" data-correct="no">
  <p class="quiz-question">Výnimky vyhadzujeme (vytvárame) pomocou kľúčového slova <code>catch</code>.</p>
  <div class="quiz-actions">
    <button type="button" class="quiz-btn quiz-yes">Áno</button>
    <button type="button" class="quiz-btn quiz-no">Nie</button>
  </div>
  <p class="quiz-comment"><strong>Nie.</strong> Na vyhodenie výnimky slúži kľúčové slovo <code>throw</code>, napríklad <code>throw IllegalArgumentException("...")</code>. <code>catch</code> naopak výnimky odchytáva.</p>
</div>

<div class="quiz-item" data-correct="yes">
  <p class="quiz-question">Vlastnú výnimku vytvoríme jednoducho tak, že vytvoríme triedu, ktorá dedí z triedy <code>Exception</code> (alebo jej podtriedy).</p>
  <div class="quiz-actions">
    <button type="button" class="quiz-btn quiz-yes">Áno</button>
    <button type="button" class="quiz-btn quiz-no">Nie</button>
  </div>
  <p class="quiz-comment"><strong>Áno.</strong> Napríklad <code>class InsufficientFundsException(amount: Double) : Exception("...")</code>.</p>
</div>

<div class="quiz-item" data-correct="no">
  <p class="quiz-question">Trieda <code>Error</code> reprezentuje bežné chyby aplikácie, ktoré by sme mali v kóde vždy odchytávať.</p>
  <div class="quiz-actions">
    <button type="button" class="quiz-btn quiz-yes">Áno</button>
    <button type="button" class="quiz-btn quiz-no">Nie</button>
  </div>
  <p class="quiz-comment"><strong>Nie.</strong> <code>Error</code> reprezentuje kritické chyby JVM (napr. nedostatok pamäte), ktoré nemá zmysel odchytávať. Bežné chyby reprezentuje trieda <code>Exception</code>.</p>
</div>

<div class="quiz-item" data-correct="yes">
  <p class="quiz-question"><code>RuntimeException</code> je podtriedou <code>Exception</code> a patria do nej napríklad <code>IllegalArgumentException</code> aj <code>NumberFormatException</code>.</p>
  <div class="quiz-actions">
    <button type="button" class="quiz-btn quiz-yes">Áno</button>
    <button type="button" class="quiz-btn quiz-no">Nie</button>
  </div>
  <p class="quiz-comment"><strong>Áno.</strong> Dedičská hierarchia je <code>Throwable → Exception → RuntimeException → IllegalArgumentException / NumberFormatException / ...</code>.</p>
</div>

<div class="quiz-item" data-correct="no">
  <p class="quiz-question">V Kotline musíme pri deklarácii funkcie explicitne uviesť, aké výnimky môže vyhodiť (podobne ako <code>throws</code> v Jave).</p>
  <div class="quiz-actions">
    <button type="button" class="quiz-btn quiz-yes">Áno</button>
    <button type="button" class="quiz-btn quiz-no">Nie</button>
  </div>
  <p class="quiz-comment"><strong>Nie.</strong> V Kotline sú všetky výnimky unchecked - na rozdiel od Javy nemusíme deklarovať, aké výnimky funkcia môže vyhodiť.</p>
</div>

<div class="quiz-item" data-correct="yes">
  <p class="quiz-question">Ak unchecked výnimku nikto neodchytí, program spadne (skončí s chybou).</p>
  <div class="quiz-actions">
    <button type="button" class="quiz-btn quiz-yes">Áno</button>
    <button type="button" class="quiz-btn quiz-no">Nie</button>
  </div>
  <p class="quiz-comment"><strong>Áno.</strong> To, že sú výnimky unchecked, neznamená, že ich netreba riešiť - iba to, že ich odchytenie nie je kompilátorom vynucované.</p>
</div>

<div class="quiz-item" data-correct="yes">
  <p class="quiz-question">Vlastná výnimka môže mať vlastnú popisnú správu odovzdanú do konštruktora rodičovskej triedy <code>Exception</code>.</p>

<div class="quiz-item" data-correct="no">

<script>
  (function () {
    const questionsWrap = document.getElementById('quiz-questions');
    const summaryEl = document.getElementById('quiz-summary');
    const scoreEl = document.getElementById('quiz-score');
    const restartBtn = document.getElementById('quiz-restart');
    const items = Array.from(document.querySelectorAll('#quiz-container .quiz-item'));
    const progress = document.getElementById('quiz-progress');
    const prevBtn = document.getElementById('quiz-prev');
    const nextBtn = document.getElementById('quiz-next');
    let current = 0;

    function updateNav() {
      const isLast = current === items.length - 1;
      prevBtn.disabled = current === 0;
      nextBtn.disabled = !items[current].classList.contains('answered');
      nextBtn.textContent = isLast ? 'Vyhodnotiť →' : 'Ďalšia →';
    }

    function showItem(index) {
      items.forEach((item, i) => item.classList.toggle('active', i === index));
      progress.textContent = 'Otázka ' + (index + 1) + ' / ' + items.length;
      updateNav();
    }

    function answerItem(item, btn, correct) {
      if (item.classList.contains('answered')) return;
      item.classList.add('answered');

      const chosen = btn.classList.contains('quiz-yes') ? 'yes' : 'no';
      const isCorrect = chosen === correct;
      item.dataset.result = isCorrect ? 'correct' : 'incorrect';
      btn.classList.add(isCorrect ? 'correct' : 'incorrect');

      if (!isCorrect) {
        const correctBtn = item.querySelector(correct === 'yes' ? '.quiz-yes' : '.quiz-no');
        correctBtn.classList.add('correct');
      }

      item.querySelectorAll('.quiz-btn').forEach((b) => { b.disabled = true; });
      item.querySelector('.quiz-comment').classList.add('visible');
      updateNav();
    }

    items.forEach((item) => {
      const correct = item.dataset.correct;
      item.querySelectorAll('.quiz-btn').forEach((btn) => {
        btn.addEventListener('click', () => answerItem(item, btn, correct));
      });
    });

    function showSummary() {
      const score = items.filter((item) => item.dataset.result === 'correct').length;
      scoreEl.textContent = score + ' / ' + items.length;
      questionsWrap.style.display = 'none';
      summaryEl.classList.add('visible');
    }

    function resetQuiz() {
      items.forEach((item) => {
        item.classList.remove('answered');
        delete item.dataset.result;
        item.querySelectorAll('.quiz-btn').forEach((b) => {
          b.disabled = false;
          b.classList.remove('correct', 'incorrect');
        });
        item.querySelector('.quiz-comment').classList.remove('visible');
      });
      summaryEl.classList.remove('visible');
      questionsWrap.style.display = '';
      current = 0;
      showItem(current);
    }

    prevBtn.addEventListener('click', () => {
      if (current > 0) {
        current -= 1;
        showItem(current);
      }
    });

    nextBtn.addEventListener('click', () => {
      if (current === items.length - 1) {
        showSummary();
        return;
      }
      current += 1;
      showItem(current);
    });

    restartBtn.addEventListener('click', resetQuiz);

    showItem(current);
  })();
</script>
